TYPICAL USAGE (v3):

  from connector.cdc_monitor import CDCMonitor

  # One-time setup
  monitor = CDCMonitor(db_url, tenant_id=1)
  monitor.setup_cdc_for_tables(["patients", "encounters"])

  # Per connection (in connection pool setup)
  @event.listens_for(Engine, "connect")
  def set_tenant(dbapi_conn, record):
      cursor = dbapi_conn.cursor()
      cursor.execute("SET app.tenant_id = 1")
      cursor.close()

  # Sync loop
  last_id = monitor.load_watermark(tenant_id=1)
  for batch in monitor.iter_changes_since(last_id, tenant_id=1, batch_size=500):
      try:
          process(batch)  # Your ETL logic
          monitor.advance_watermark(tenant_id=1, batch[-1]["change_id"])
      except Exception as e:
          log.error(f"Batch failed: {e}")
          break  # Watermark unchanged, retry next cycle
